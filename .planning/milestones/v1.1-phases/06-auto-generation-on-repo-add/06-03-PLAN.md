Add IPC event wiring inside the component (before the return):

```typescript
const { setProgress, setComplete, setCancelled, dismissJob } = useGenerationQueueStore();
const { addToast } = useToastStore();

useEffect(() => {
  // Subscribe to generation progress events
  const unsubProgress = window.reef.c4Generation.onProgress((_event, data) => {
    setProgress(data);
  });

  // Subscribe to generation completion events
  const unsubComplete = window.reef.c4Generation.onComplete((_event, data) => {
    setComplete(data);

    if (data.success) {
      addToast({
        type: 'success',
        message: `Diagrams ready for ${data.repoName}`,
        duration: 5000, // Auto-dismiss after 5 seconds
      });
      // Auto-dismiss the completed job from status bar after a short delay
      setTimeout(() => dismissJob(data.repoPath), 2000);
    } else {
      // Partial failure: show which levels succeeded
      const partialMsg = data.completedLevels.length > 0
        ? ` (${data.completedLevels.join(', ')} levels completed)`
        : '';
      addToast({
        type: 'error',
        message: `Generation failed for ${data.repoName}${partialMsg}`,
        duration: 0, // Error toasts do NOT auto-dismiss
        action: {
          label: 'Retry',
          onClick: () => {
            // Re-enqueue the repo for generation
            const { addJob } = useGenerationQueueStore.getState();
            addJob(data.repoPath, data.repoName);
            window.reef.c4Generation.enqueue(data.repoPath, data.repoName);
          },
        },
      });
    }
  });

  // Subscribe to cancellation events
  const unsubCancelled = window.reef.c4Generation.onCancelled((_event, data) => {
    setCancelled(data.repoPath);
    addToast({
      type: 'info',
      message: `Generation cancelled for ${data.repoName}`,
      duration: 3000,
    });
    // Remove from status bar
    setTimeout(() => dismissJob(data.repoPath), 1000);
  });

  return () => {
    unsubProgress();
    unsubComplete();
    unsubCancelled();
  };
}, []);
```

**CRITICAL: The `useEffect` dependency array is intentionally empty `[]`.** The store functions (`setProgress`, `setComplete`, etc.) are stable references from Zustand -- they don't change between renders. Including them would cause re-subscriptions on every render.

**HOWEVER**: To satisfy exhaustive-deps lint rules without causing re-subscriptions, destructure from `useGenerationQueueStore.getState()` INSIDE the effect instead of using the hook destructuring. Alternatively, access the store methods via `useGenerationQueueStore.getState().setProgress(data)` inside the callbacks. This is the safest pattern:

```typescript
useEffect(() => {
  const unsubProgress = window.reef.c4Generation.onProgress((_event, data) => {
    useGenerationQueueStore.getState().setProgress(data);
  });

  const unsubComplete = window.reef.c4Generation.onComplete((_event, data) => {
    useGenerationQueueStore.getState().setComplete(data);
    const { addToast: toast } = useToastStore.getState();

    if (data.success) {
      toast({ type: 'success', message: `Diagrams ready for ${data.repoName}`, duration: 5000 });
      setTimeout(() => useGenerationQueueStore.getState().dismissJob(data.repoPath), 2000);
    } else {
      const partialMsg = data.completedLevels.length > 0
        ? ` (${data.completedLevels.join(', ')} levels completed)`
        : '';
      toast({
        type: 'error',
        message: `Generation failed for ${data.repoName}${partialMsg}`,
        duration: 0,
        action: {
          label: 'Retry',
          onClick: () => {
            useGenerationQueueStore.getState().addJob(data.repoPath, data.repoName);
            window.reef.c4Generation.enqueue(data.repoPath, data.repoName);
          },
        },
      });
    }
  });

  const unsubCancelled = window.reef.c4Generation.onCancelled((_event, data) => {
    useGenerationQueueStore.getState().setCancelled(data.repoPath);
    useToastStore.getState().addToast({ type: 'info', message: `Generation cancelled for ${data.repoName}`, duration: 3000 });
    setTimeout(() => useGenerationQueueStore.getState().dismissJob(data.repoPath), 1000);
  });

  return () => {
    unsubProgress();
    unsubComplete();
    unsubCancelled();
  };
}, []);
```
