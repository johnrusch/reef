        styleEl.setAttribute('data-reef-changes', '');
        styleEl.textContent = `
          [data-changed="direct"] rect,
          [data-changed="direct"] polygon {
            fill: rgba(251, 191, 36, 0.25) !important;
            stroke: rgb(245, 158, 11) !important;
