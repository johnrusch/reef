```typescript
const handleSvgClick = useCallback((event: React.MouseEvent<HTMLDivElement>) => {
  // Only process if click detection is enabled and callback provided
  if (!onElementClick || !isClickable) return;

  const target = event.target as SVGElement;

  // Traverse up DOM tree to find element with ID
  // PlantUML generates elements like <g id="elem_reef_main">
  // Or directly on shapes like <rect id="reef_main">
  let element: Element | null = target;
  let elementId: string | null = null;

  while (element && element !== event.currentTarget) {
    // Check for ID attribute
    const id = element.getAttribute('id');

    if (id) {
      // Skip SVG root and internal PlantUML IDs
      if (id === 'svg_root' || id.startsWith('_')) {
        element = element.parentElement;
        continue;
      }

      // PlantUML wraps elements in groups with "elem_" prefix
      if (id.startsWith('elem_')) {
        elementId = id.replace('elem_', '');
        break;
      }

      // Direct element ID (for some diagram types)
      elementId = id;
      break;
    }

    element = element.parentElement;
  }

  if (elementId) {
    console.log('Clicked element:', elementId);
    onElementClick(elementId);
  }
}, [onElementClick, isClickable]);
```

```typescript
const handleElementClick = useCallback(async (elementId: string) => {
  // Don't process clicks during generation or for non-C4 diagrams
  if (isGenerating || !currentOptions.type.startsWith('c4-')) return;

  const currentNavLevel = navigationStore.currentLevel();
  const nextLevel = getNextLevel(currentNavLevel.level);

  // Can't drill down from Code level
  if (!nextLevel) {
    console.log('Already at Code level, cannot drill down further');
    return;
  }

  // Determine human-readable name for breadcrumb
  // The elementId is sanitized (e.g., "reef_main"), convert back to readable
  const elementName = elementId
    .replace(/_/g, ' ')
    .replace(/\b\w/g, l => l.toUpperCase());

  console.log(`Drilling down to ${nextLevel}: ${elementId} (${elementName})`);

  // Push new level to navigation stack
  navigationStore.push({
    level: nextLevel,
    elementId: elementId,
    elementName: elementName,
  });

  // Update diagram type and regenerate
  const newType = `c4-${nextLevel}` as DiagramType;
  handleControlChange({ type: newType });

  try {
    await onRegenerateDiagram({
      ...currentOptions,
      type: newType,
      elementId: elementId,
    });
  } catch (error) {
    // Revert navigation on error
    navigationStore.pop();
    console.error('Failed to drill down:', error);
  }
}, [isGenerating, currentOptions, navigationStore, onRegenerateDiagram, handleControlChange]);
```

Add CSS rules for hover effects on clickable SVG elements. These provide visual feedback that elements can be clicked.

Add to the end of index.css (after existing styles):

```css
/* C4 Diagram Click Navigation
 * Visual indicators for clickable diagram elements
 * PlantUML generates SVG with elements having IDs like "elem_reef_main"
 */

/* Base styles for clickable diagram containers */
.diagram-wrapper.clickable {
  /* No global cursor - handled per-element */
}

/* Target elements with IDs (PlantUML generated) */
.diagram-wrapper.clickable svg [id]:not([id^="_"]):not([id="svg_root"]) {
  cursor: pointer;
  transition: opacity 0.15s ease, filter 0.15s ease;
}

/* Hover effect - subtle brightness increase and opacity change */
.diagram-wrapper.clickable svg [id]:not([id^="_"]):not([id="svg_root"]):hover {
  opacity: 0.85;
  filter: brightness(1.15);
}

/* Specifically target C4 element groups (PlantUML wraps in <g> with elem_ prefix) */
.diagram-wrapper.clickable svg g[id^="elem_"] {
  cursor: pointer;
}

.diagram-wrapper.clickable svg g[id^="elem_"]:hover rect,
.diagram-wrapper.clickable svg g[id^="elem_"]:hover path,
.diagram-wrapper.clickable svg g[id^="elem_"]:hover polygon {
  filter: brightness(1.1);
}

/* Text elements within clickable groups should also show pointer */
.diagram-wrapper.clickable svg g[id^="elem_"] text {
  cursor: pointer;
}

/* Disable hover effects during diagram generation */
.diagram-wrapper:not(.clickable) svg [id] {
  cursor: default;
}

/* Focus indicator for keyboard navigation (Phase 4) */
.diagram-wrapper.clickable svg [id]:focus-visible {
  outline: 2px solid #3b82f6;
  outline-offset: 2px;
}
```

Key considerations:
- Use :not() selectors to exclude internal PlantUML IDs (starting with underscore)
- Target both direct IDs and elem_ prefixed groups
- Subtle brightness increase (1.1-1.15) rather than drastic color change
- Opacity change provides another visual cue
- Transition for smooth effect
- Focus-visible for future keyboard navigation
