    After the context and container levels generate successfully, query the stored container diagram to extract element IDs. PlantUML C4 container diagrams contain element definitions like `Container(element_id, "Name", ...)`. Parse these to get elementIds for component generation.

    Add a helper function inside `generationQueueService.ts`:

    ```typescript
    function extractElementIds(diagramContent: string, level: C4Level): string[] {
      // C4 PlantUML element patterns by level:
      // Container diagrams have: Container(id, "name", ...) or ContainerDb(id, ...) etc.
      // Component diagrams have: Component(id, "name", ...) etc.
      const patterns: Record<string, RegExp> = {
        container: /(?:Container|ContainerDb|ContainerQueue|Container_Ext|Container_Boundary)\((\w+)/g,
        component: /(?:Component|ComponentDb|ComponentQueue|Component_Ext)\((\w+)/g,
      };
      const regex = patterns[level];
      if (!regex) return [];
      const ids: string[] = [];
      let match: RegExpExecArray | null;
      while ((match = regex.exec(diagramContent)) !== null) {
        ids.push(match[1]);
      }
      return ids;
    }
    ```

      try {

            completedLevels.push(`code:${componentId}`);
          } catch (err) {

      }
    }

    // Subscribe to generation queue events for Generate All progress
    useEffect(() => {
      if (!repository) return;

      const unsubProgress = window.reef.c4Generation.onProgress((_, data) => {
