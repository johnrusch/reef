Create IPC handlers that expose C4StorageService and MigrationService to renderer:

1. **Imports**:
   - ipcMain, BrowserWindow from 'electron'
   - C4StorageService from './c4StorageService'
   - MigrationService from './migrationService'
   - DiagramState, StoredDiagram from '@shared/types/diagramState'

2. **Singleton instances**:
   ```typescript
   let storageService: C4StorageService | null = null;
   let migrationService: MigrationService | null = null;

   const getStorageService = (): C4StorageService => {
     if (!storageService) {
       storageService = new C4StorageService();
     }
     return storageService;
   };
   ```

3. **registerC4StorageHandlers() function**:

   **Initialization handler** (runs migration if needed):
   ```typescript
   ipcMain.handle('c4-storage:initialize', async () => {
     const migration = new MigrationService();

     if (migration.needsMigration()) {
       try {
         migration.migrate();
         migration.cleanupV1Cache();
       } catch (error) {
         console.error('Migration failed:', error);
         // Continue anyway - diagrams will be marked as never_generated
       }
     }

     return { initialized: true };
   });
   ```

   **Get diagram**:
   ```typescript
   ipcMain.handle('c4-storage:get-diagram', async (_, repoPath: string, level: string, elementId?: string) => {
     return getStorageService().getDiagram(repoPath, level as C4Level, elementId);
   });
   ```

   **Store diagram**:
   ```typescript
   ipcMain.handle('c4-storage:store-diagram', async (_, diagram: StoredDiagram) => {
     getStorageService().storeDiagram(diagram);
     return { success: true };
   });
   ```

   **Update state**:
   ```typescript
   ipcMain.handle('c4-storage:update-state', async (_, repoPath: string, level: string, state: DiagramState, elementId?: string, errorMessage?: string) => {
     getStorageService().updateState(repoPath, level as C4Level, state, elementId, errorMessage);

     // Notify all windows of state change
     BrowserWindow.getAllWindows().forEach(win => {
       win.webContents.send('c4-storage:state-changed', {
         repoPath,
         level,
         elementId,
         state,
         errorMessage,
       });
     });

     return { success: true };
   });
   ```

   **Get state**:
   ```typescript
   ipcMain.handle('c4-storage:get-state', async (_, repoPath: string, level: string, elementId?: string) => {
     return getStorageService().getState(repoPath, level as C4Level, elementId);
   });
   ```

   **Get all states for repo**:
   ```typescript
   ipcMain.handle('c4-storage:get-repo-states', async (_, repoPath: string) => {
     const diagrams = getStorageService().getAllDiagramsForRepo(repoPath);
     return diagrams.map(d => ({
       repoPath: d.repoPath,
       level: d.level,
       elementId: d.elementId,
       state: d.state,
       errorMessage: d.errorMessage,
     }));
   });
   ```

   **Get storage stats** (for Settings display):
   ```typescript
   ipcMain.handle('c4-storage:get-stats', async () => {
     return getStorageService().getStorageStats();
   });
   ```

   **Clear all diagrams** (for Settings "Clear All" button):
   ```typescript
   ipcMain.handle('c4-storage:clear-all', async () => {
     getStorageService().clearAllDiagrams();
     return { success: true };
   });
   ```

4. **Cleanup handler**:
   ```typescript
   const cleanupC4Storage = () => {
     if (storageService) {
       storageService.close();
       storageService = null;
     }
   };
   ```

5. **Export** registerC4StorageHandlers and cleanupC4Storage

4. **Subscribe to state changes from main process**:
   ```typescript
   useEffect(() => {
     const unsubscribe = window.reef.c4Storage.onStateChanged((_, data) => {
       if (data.repoPath === repoPath) {
         setState(data.repoPath, data.level, data.state, data.elementId, data.errorMessage);
       }
     });

     return unsubscribe;
   }, [repoPath, setState]);
   ```

7. **Update handleGenerate/handleRegenerate** to update state:
   ```typescript
   const handleGenerate = async () => {
     // Update state to generating
     await window.reef.c4Storage.updateState(repoPath, currentLevel, 'generating', elementId);

     try {
       // Call existing generation logic
       const result = await generateDiagram(repoPath, currentLevel, elementId);

       // Store diagram and update state to fresh
       await window.reef.c4Storage.storeDiagram({
         repoPath,
         level: currentLevel,
         elementId,
         diagramContent: result.content,
         state: 'fresh',
         modelUsed: result.modelUsed,
         promptVersion: result.promptVersion,
         tokensUsed: result.tokensUsed,
         generationCost: result.cost,
       });
     } catch (error) {
       // Update state to error
       await window.reef.c4Storage.updateState(
         repoPath,
         currentLevel,
         'error',
         elementId,
         'Could not generate diagram. Please try again.'
       );
     }
   };
   ```
