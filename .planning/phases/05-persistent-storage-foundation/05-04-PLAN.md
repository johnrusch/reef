   import { useDiagramStateStore } from '@renderer/stores/diagramStateStore';
   import { renderHook, act } from '@testing-library/react';

   describe('diagramStateStore', () => {
     beforeEach(() => {
       // Reset store between tests
       useDiagramStateStore.setState({ states: new Map() });
     });

     describe('getState', () => {
       it('returns never_generated for unknown entries', () => {
         const { result } = renderHook(() => useDiagramStateStore());
         expect(result.current.getState('/repo', 'context')).toBe('never_generated');
       });
     });

     describe('setState', () => {
       it('sets state and creates entry', () => {
         const { result } = renderHook(() => useDiagramStateStore());

         act(() => {
           result.current.setState('/repo', 'context', 'fresh');
         });

         expect(result.current.getState('/repo', 'context')).toBe('fresh');
       });
     });

     describe('transitions', () => {
       it('transitionToGenerating sets generating state', () => {
         const { result } = renderHook(() => useDiagramStateStore());

         act(() => {
           result.current.transitionToGenerating('/repo', 'context');
         });

         expect(result.current.getState('/repo', 'context')).toBe('generating');
       });

       it('transitionToError stores error message', () => {
         const { result } = renderHook(() => useDiagramStateStore());

         act(() => {
           result.current.transitionToError('/repo', 'context', 'Test error');
         });

         expect(result.current.getState('/repo', 'context')).toBe('error');
         expect(result.current.getEntry('/repo', 'context')?.errorMessage).toBe('Test error');
       });
     });

     describe('loadStatesFromBackend', () => {
       it('replaces all states with backend data', () => {
         const { result } = renderHook(() => useDiagramStateStore());

         act(() => {
           result.current.loadStatesFromBackend([
             { repoPath: '/repo1', level: 'context', state: 'fresh' },
             { repoPath: '/repo2', level: 'container', state: 'stale' },
           ]);
         });

         expect(result.current.getState('/repo1', 'context')).toBe('fresh');
         expect(result.current.getState('/repo2', 'container')).toBe('stale');
       });
     });
   });
